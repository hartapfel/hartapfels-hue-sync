cmake_minimum_required(VERSION 3.18)
project(C9ScreenSync LANGUAGES C)
add_library(c9_bitmap SHARED src/bitmap.c src/color.c src/video.c)
target_include_directories(c9_bitmap PUBLIC src)
set_target_properties(c9_bitmap PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON C_STANDARD 11)
if(NOT MSVC)
  target_link_libraries(c9_bitmap PRIVATE m)
endif()

set(MBEDTLS_SOURCE_DIR "" CACHE PATH "Unpacked official Mbed TLS 3.6 release")
if(MBEDTLS_SOURCE_DIR)
  set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
  set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
  set(CMAKE_POSITION_INDEPENDENT_CODE ON)
  add_subdirectory("${MBEDTLS_SOURCE_DIR}" mbedtls EXCLUDE_FROM_ALL)
  add_library(c9_hue_stream SHARED src/hue_stream.c)
  target_include_directories(c9_hue_stream PUBLIC src)
  target_link_libraries(c9_hue_stream PRIVATE mbedtls)
  set_target_properties(c9_hue_stream PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON C_STANDARD 11)
endif()

set(PICCAP_SOURCE_DIR "" CACHE PATH "hyperion-webos source tree providing the video backend ABI headers")
if(PICCAP_SOURCE_DIR AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(c9-capture-check src/capture_check.c src/capture.c src/bitmap.c src/color.c src/video.c)
  target_include_directories(c9-capture-check PRIVATE
    "${PICCAP_SOURCE_DIR}/unicapture" "${PICCAP_SOURCE_DIR}/src")
  target_link_libraries(c9-capture-check PRIVATE m ${CMAKE_DL_LIBS})
  set_target_properties(c9-capture-check PROPERTIES ENABLE_EXPORTS ON C_STANDARD 11)
endif()

option(C9_WORKER_TEST "Build the worker with a generated frame source for loopback testing" OFF)
if(TARGET c9_hue_stream AND ((PICCAP_SOURCE_DIR AND CMAKE_SYSTEM_NAME STREQUAL "Linux") OR C9_WORKER_TEST))
  enable_language(CXX)
  find_package(Threads REQUIRED)
  set(worker_sources src/worker.cpp src/bitmap.c src/color.c src/video.c)
  if(C9_WORKER_TEST)
    add_executable(c9-sync-fixture ${worker_sources} tests/worker_capture.cpp)
    set(worker_target c9-sync-fixture)
  else()
    add_executable(c9-sync ${worker_sources} src/capture.c)
    set(worker_target c9-sync)
    target_include_directories(c9-sync PRIVATE "${PICCAP_SOURCE_DIR}/unicapture" "${PICCAP_SOURCE_DIR}/src")
    set_target_properties(c9-sync PROPERTIES ENABLE_EXPORTS ON BUILD_WITH_INSTALL_RPATH ON INSTALL_RPATH "$ORIGIN")
    target_link_libraries(c9-sync PRIVATE m ${CMAKE_DL_LIBS})
  endif()
  target_include_directories(${worker_target} PRIVATE src)
  target_compile_features(${worker_target} PRIVATE cxx_std_11)
  target_link_libraries(${worker_target} PRIVATE c9_hue_stream Threads::Threads)
endif()
