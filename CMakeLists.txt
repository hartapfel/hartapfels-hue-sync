cmake_minimum_required(VERSION 3.18)
project(C9ScreenSync LANGUAGES C)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_program(HUE_SYNC_DESKTOP NAMES HueSync.exe HINTS "$ENV{ProgramFiles}/Hue Sync"
  REQUIRED NO_CMAKE_FIND_ROOT_PATH)
find_program(LLVM_OBJDUMP NAMES llvm-objdump HINTS "$ENV{ProgramFiles}/LLVM/bin"
  REQUIRED NO_CMAKE_FIND_ROOT_PATH)
set(color_generator "${CMAKE_CURRENT_SOURCE_DIR}/cmake/desktop_color.py")
set(desktop_color "${CMAKE_CURRENT_BINARY_DIR}/desktop_color.c")
add_custom_command(OUTPUT "${desktop_color}"
  COMMAND "${Python3_EXECUTABLE}" "${color_generator}" "${HUE_SYNC_DESKTOP}" "${LLVM_OBJDUMP}" "${desktop_color}"
  DEPENDS "${color_generator}" "${HUE_SYNC_DESKTOP}" "${LLVM_OBJDUMP}" VERBATIM)
add_library(c9_desktop_color OBJECT "${desktop_color}")
target_include_directories(c9_desktop_color PRIVATE src)
set_target_properties(c9_desktop_color PROPERTIES C_STANDARD 11 POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(c9_desktop_color PRIVATE /fp:strict)
else()
  target_compile_options(c9_desktop_color PRIVATE -ffp-contract=off)
endif()
set(processing_sources src/bitmap.c src/blackbars.c src/color.c src/video.c $<TARGET_OBJECTS:c9_desktop_color>)
add_library(c9_bitmap SHARED ${processing_sources})
target_include_directories(c9_bitmap PUBLIC src)
set_target_properties(c9_bitmap PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON C_STANDARD 11)
if(NOT MSVC)
  target_link_libraries(c9_bitmap PRIVATE m)
endif()

set(MBEDTLS_SOURCE_DIR "" CACHE PATH "Unpacked official Mbed TLS 3.6 release")
if(MBEDTLS_SOURCE_DIR)
  set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
  set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
  set(CMAKE_POSITION_INDEPENDENT_CODE ON)
  add_subdirectory("${MBEDTLS_SOURCE_DIR}" mbedtls EXCLUDE_FROM_ALL)
  add_library(c9_hue_stream SHARED src/hue_stream.c)
  target_include_directories(c9_hue_stream PUBLIC src)
  target_link_libraries(c9_hue_stream PRIVATE mbedtls)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_link_libraries(c9_hue_stream PRIVATE rt)
  endif()
  set_target_properties(c9_hue_stream PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON C_STANDARD 11)
endif()

set(PICCAP_SOURCE_DIR "" CACHE PATH "hyperion-webos source tree providing the video backend ABI headers")
if(PICCAP_SOURCE_DIR AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_library(c9-test-capture-backend SHARED EXCLUDE_FROM_ALL tests/direct_backend.c)
  target_include_directories(c9-test-capture-backend PRIVATE src
    "${PICCAP_SOURCE_DIR}/unicapture" "${PICCAP_SOURCE_DIR}/src")
  add_executable(c9-capture-zones src/capture_zones.c src/music.c src/capture.c ${processing_sources})
  target_include_directories(c9-capture-zones PRIVATE src "${PICCAP_SOURCE_DIR}/unicapture" "${PICCAP_SOURCE_DIR}/src")
  target_link_libraries(c9-capture-zones PRIVATE m rt ${CMAKE_DL_LIBS})
  set_target_properties(c9-capture-zones PROPERTIES ENABLE_EXPORTS ON C_STANDARD 11)
  add_executable(c9-capture-check src/capture_check.c src/capture.c ${processing_sources})
  target_include_directories(c9-capture-check PRIVATE src
    "${PICCAP_SOURCE_DIR}/unicapture" "${PICCAP_SOURCE_DIR}/src")
  target_link_libraries(c9-capture-check PRIVATE m rt ${CMAKE_DL_LIBS})
  set_target_properties(c9-capture-check PROPERTIES ENABLE_EXPORTS ON C_STANDARD 11)
endif()

option(C9_WORKER_TEST "Build the worker with a generated frame source for loopback testing" OFF)
if(TARGET c9_hue_stream AND ((PICCAP_SOURCE_DIR AND CMAKE_SYSTEM_NAME STREQUAL "Linux") OR C9_WORKER_TEST))
  enable_language(CXX)
  find_package(Threads REQUIRED)
  find_path(FLATBUFFERS_INCLUDE_DIR flatbuffers/flatbuffers.h REQUIRED)
  find_program(FLATC_EXECUTABLE flatc REQUIRED NO_CMAKE_FIND_ROOT_PATH)
  set(protocol_headers)
  foreach(protocol IN ITEMS hyperion_request hyperion_reply)
    set(schema "${PICCAP_SOURCE_DIR}/fbs/${protocol}.fbs")
    set(header "${CMAKE_CURRENT_BINARY_DIR}/piccap/${protocol}_generated.h")
    add_custom_command(OUTPUT "${header}"
      COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/piccap"
      COMMAND "${FLATC_EXECUTABLE}" --cpp -o "${CMAKE_CURRENT_BINARY_DIR}/piccap" "${schema}"
      DEPENDS "${schema}" "${FLATC_EXECUTABLE}" VERBATIM)
    list(APPEND protocol_headers "${header}")
  endforeach()
  set(worker_sources src/worker.cpp src/piccap.cpp src/chart_probe.cpp ${protocol_headers} ${processing_sources})
  if(C9_WORKER_TEST)
    add_executable(c9-sync-fixture ${worker_sources} tests/worker_capture.cpp)
    target_compile_definitions(c9-sync-fixture PRIVATE C9_WORKER_TEST)
    set(worker_target c9-sync-fixture)
  else()
    add_executable(c9-sync ${worker_sources} src/capture.c)
    set(worker_target c9-sync)
    target_include_directories(c9-sync PRIVATE "${PICCAP_SOURCE_DIR}/unicapture" "${PICCAP_SOURCE_DIR}/src")
    set_target_properties(c9-sync PROPERTIES ENABLE_EXPORTS ON BUILD_WITH_INSTALL_RPATH ON INSTALL_RPATH "$ORIGIN")
    target_link_libraries(c9-sync PRIVATE m rt ${CMAKE_DL_LIBS})
  endif()
  target_include_directories(${worker_target} PRIVATE src "${CMAKE_CURRENT_BINARY_DIR}/piccap" "${FLATBUFFERS_INCLUDE_DIR}")
  target_compile_features(${worker_target} PRIVATE cxx_std_11)
  target_link_libraries(${worker_target} PRIVATE c9_hue_stream Threads::Threads mbedtls)
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR STREQUAL "arm")
  set(relocation_script "${CMAKE_CURRENT_SOURCE_DIR}/cmake/webos-relocations.ld")
  foreach(target IN ITEMS c9-sync c9_hue_stream c9-capture-check c9-capture-zones c9-test-capture-backend)
    if(TARGET ${target})
      target_link_options(${target} PRIVATE "LINKER:-T,${relocation_script}")
      set_property(TARGET ${target} APPEND PROPERTY LINK_DEPENDS "${relocation_script}")
    endif()
  endforeach()
endif()
